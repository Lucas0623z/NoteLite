cmake_minimum_required(VERSION 3.16)
project(NoteLiteAudio LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "NoteLiteAudio capture helper currently targets Windows only")
endif()
add_executable(NoteLiteAudio main.c pitch.c)
target_compile_features(NoteLiteAudio PRIVATE c_std_11)
if(MSVC)
  target_compile_options(NoteLiteAudio PRIVATE /O2 /W4)
else()
  target_compile_options(NoteLiteAudio PRIVATE -O2 -Wall -Wextra)
  target_link_options(NoteLiteAudio PRIVATE -municode -static-libgcc)
  target_link_libraries(NoteLiteAudio PRIVATE m)
endif()
